\chapter{Full Proof of IRLS's Local Convergence}
\label{app:irls-local-conv-proof}
This appendix provides a step by step derivation of the local linear convergence rate of the IRLS algorithm as stated in the main report.

\section{The Smoothed Objective and Its Fixed-Point Map}
\label{app:sec:smooth}

\paragraph{Huber smoothing of the penalty.}
The safety threshold $\varepsilon$ (written $\varepsilon_{\mathrm{thr}}$ in the main text) introduced in the weights formula prevents division by zero. However, it also changes the objective~\eqref{eq:obj_fun} into a smoothed one:
\begin{equation}
    f_\varepsilon(\mathbf{w})
    = \frac{1}{2}\|\mathbf{X}\mathbf{w}-\mathbf{y}\|_2^2
    + \lambda\sum_{i=1}^{H}\psi_\varepsilon(w_i)
    \label{app:eq:smooth_obj}
\end{equation}
where $\psi_\varepsilon(t)$ is defined as:
\begin{equation}
    \psi_\varepsilon(t)
    =\begin{cases}
        |t| & |t|>\varepsilon\\
        \frac{t^2}{2\varepsilon}+\frac{\varepsilon}{2} & |t|\le\varepsilon
    \end{cases}
    \label{app:eq:huber}
\end{equation}
This is a Huber smoothing of the absolute value. It agrees with $|t|$ on the \emph{active set} $\mathcal{A}=\{i:|(w^*)_i|>\varepsilon\}$ and replaces it with a quadratic curve on the inactive features \emph{pinned set} $\mathcal{P}=\{i:|(w^*)_i|<\varepsilon\}$. 
The minimiser of this smoothed objective converges to the true LASSO minimiser as $\varepsilon$ approaches zero. The final objective difference is $O(\varepsilon)$.

\paragraph{Strong convexity.}
\begin{proposition}
\label{app:prop:strong_convex}
If the data matrix $\mathbf{X}$ has full column rank, then the smoothed objective is strongly convex and has a unique minimum.
\end{proposition}
\begin{proof}
The Hessian matrix of the smoothed objective is:
\begin{equation}
    \nabla^2 f_\varepsilon(\mathbf{w})
    = \mathbf{X}^T\mathbf{X}
    + \lambda\,\mathrm{diag}\bigl(\psi_\varepsilon''(w_i)\bigr)
    \label{app:eq:hessian_fe}
\end{equation}
where the second derivative $\psi_\varepsilon''(t)$ is $0$ for $|t|>\varepsilon$ and $1/\varepsilon$ for $|t|<\varepsilon$.
Because $\mathbf{X}$ has full column rank, the product $\mathbf{X}^T\mathbf{X}$ is positive definite. The diagonal correction term is positive semidefinite. Their sum is therefore positive definite. This guarantees that the function is strongly convex and the minimum is unique.
\end{proof}

\paragraph{Fixed-point map.}
Each IRLS iteration solves the weighted normal equations
$\mathbf{Q}(\mathbf{w}_k)\,\mathbf{w}_{k+1}=\mathbf{b}$ with
$\mathbf{Q}(\mathbf{w})=\mathbf{X}^T\mathbf{X}+\lambda\,\mathrm{diag}\!\bigl(1/\max(|w_i|,\varepsilon)\bigr)$
and $\mathbf{b}=\mathbf{X}^T\mathbf{y}$ (throughout this appendix $\lambda\equiv\lambda_{\text{LASSO}}$),
so the algorithm is the fixed-point iteration $\mathbf{w}_{k+1}=\mathcal{M}(\mathbf{w}_k)$ with
$\mathcal{M}(\mathbf{w})=\mathbf{Q}(\mathbf{w})^{-1}\mathbf{b}$.
The smoothed minimiser $\mathbf{w}^*$ is a fixed point: stationarity
$\nabla f_\varepsilon(\mathbf{w}^*)=\mathbf{X}^T(\mathbf{X}\mathbf{w}^*-\mathbf{y})+\lambda\,\psi_\varepsilon'(\mathbf{w}^*)=\mathbf{0}$
coincides with $\mathbf{Q}(\mathbf{w}^*)\mathbf{w}^*=\mathbf{b}$, because the weight times the
component, $w^*_i/\max(|w^*_i|,\varepsilon)$, equals $\psi_\varepsilon'(w^*_i)$ on both the
active set ($\mathrm{sign}(w^*_i)$) and the pinned set ($w^*_i/\varepsilon$). Hence
$\mathcal{M}(\mathbf{w}^*)=\mathbf{w}^*$.

\section{Jacobian of the Fixed-Point Map}
\label{app:sec:jacobian}

We need to compute the Jacobian of the iteration map evaluated at the optimal solution.

\paragraph{Matrix differential identity.}
For any invertible matrix $\mathbf{Q}(\mathbf{w})$, the derivative of its inverse is given by $d(\mathbf{Q}^{-1}) = -\mathbf{Q}^{-1}(d\mathbf{Q})\mathbf{Q}^{-1}$.
Applying this rule to our map $\mathcal{M}(\mathbf{w})=\mathbf{Q}(\mathbf{w})^{-1}\mathbf{b}$ yields:
\begin{equation}
    d\mathcal{M}
    = -\mathbf{Q}^{-1}(d\mathbf{Q})\mathbf{Q}^{-1}\mathbf{b}
    = -\mathbf{Q}^{-1}(d\mathbf{Q})\mathbf{w}^*
    \label{app:eq:dM}
\end{equation}
In the last step we used the fixed point property $\mathbf{Q}^{-1}\mathbf{b}=\mathbf{w}^*$.

\paragraph{Evaluating the differential components.}
Let $\mathbf{B}$ be the matrix $\mathbf{Q}$ evaluated at $\mathbf{w}^*$ and let $\mathbf{H}$ be the true Hessian $\nabla^2 f_\varepsilon(\mathbf{w}^*)$.
Comparing them, note first that $\mathbf{B}$ and $\mathbf{H}$ share the same off-diagonal part $\mathbf{X}^T\mathbf{X}$ and differ only on the diagonal, so $\mathbf{B}-\mathbf{H}$ is itself diagonal.
On the active features ($|w^*_i|>\varepsilon$):
\[
    \mathbf{B}_{ii} = (\mathbf{X}^T\mathbf{X})_{ii} + \frac{\lambda}{|w^*_i|}
\]
\[
    \mathbf{H}_{ii} = (\mathbf{X}^T\mathbf{X})_{ii} + 0
\]
The difference is $\mathbf{B}_{ii} - \mathbf{H}_{ii} = \frac{\lambda}{|w^*_i|}$.
On the inactive features ($|w^*_i|<\varepsilon$):
\[
    \mathbf{B}_{ii} = (\mathbf{X}^T\mathbf{X})_{ii} + \frac{\lambda}{\varepsilon}
\]
\[
    \mathbf{H}_{ii} = (\mathbf{X}^T\mathbf{X})_{ii} + \frac{\lambda}{\varepsilon}
\]
The difference is $\mathbf{B}_{ii} - \mathbf{H}_{ii} = 0$.
To see why this difference is what governs the Jacobian, we compute $d\mathbf{Q}$ explicitly. Only the diagonal of $\mathbf{Q}(\mathbf{w})=\mathbf{X}^T\mathbf{X}+\lambda\,\mathrm{diag}(1/\max(|w_i|,\varepsilon))$ depends on $\mathbf{w}$, so $d\mathbf{Q}$ is diagonal; differentiating the $i$-th entry,
\[
    \frac{d}{dw_i}\,\frac{1}{\max(|w_i|,\varepsilon)}
    =\begin{cases}
        -\,\mathrm{sign}(w_i)/w_i^2 & |w_i|>\varepsilon \quad(\text{active}),\\[2pt]
        0 & |w_i|<\varepsilon \quad(\text{pinned: the entry is the constant }1/\varepsilon).
    \end{cases}
\]
Multiplying $d\mathbf{Q}$ by $\mathbf{w}^*$ component-wise, on the active set
\[
    (d\mathbf{Q}\,\mathbf{w}^*)_i
    = \lambda\Bigl(-\tfrac{\mathrm{sign}(w^*_i)}{(w^*_i)^2}\Bigr)w^*_i\,dw_i
    = -\frac{\lambda}{|w^*_i|}\,dw_i
    = -(\mathbf{B}-\mathbf{H})_{ii}\,dw_i,
\]
and on the pinned set $(d\mathbf{Q}\,\mathbf{w}^*)_i = 0 = -(\mathbf{B}-\mathbf{H})_{ii}\,dw_i$. In matrix form this is exactly:
\begin{equation}
    (d\mathbf{Q})\mathbf{w}^* = -(\mathbf{B}-\mathbf{H})d\mathbf{w}
    \label{app:eq:dQ_clean}
\end{equation}

\paragraph{Jacobian formula.}
We substitute this result back into the differential equation of our map:
\[
    d\mathcal{M}
    = -\mathbf{B}^{-1}\bigl[-(\mathbf{B}-\mathbf{H})\bigr]d\mathbf{w}
    = (\mathbf{I}-\mathbf{B}^{-1}\mathbf{H})d\mathbf{w}
\]
This gives us the final formula for the Jacobian of the algorithm at the minimum:
\begin{equation}
    \nabla\mathcal{M}(\mathbf{w}^*) = \mathbf{I} - \mathbf{B}^{-1}\mathbf{H}
    \label{app:eq:jacobian}
\end{equation}

\section{Spectral Radius Bound}
\label{app:sec:spectral}

We now prove that the spectral radius of this Jacobian is strictly less than 1. We already established that both $\mathbf{B}$ and $\mathbf{H}$ are symmetric positive definite matrices.

\subsection{Step 1: Eigenvalues are real and positive}
The product $\mathbf{B}^{-1}\mathbf{H}$ is not symmetric, so its eigenvalues might be complex. To solve this, we define a new matrix $\mathbf{S} = \mathbf{B}^{-1/2}\mathbf{H}\mathbf{B}^{-1/2}$.

\begin{lemma}[Similarity of $\mathbf{B}^{-1}\mathbf{H}$ and $\mathbf{S}$]
\label{app:lem:similar}
$\mathbf{B}^{-1}\mathbf{H}$ and $\mathbf{S}$ are similar matrices, so they share the same
eigenvalues.
\end{lemma}
\begin{proof}
Two matrices $\mathbf{C}$ and $\mathbf{D}$ are similar if there exists an invertible $\mathbf{P}$
such that $\mathbf{D}=\mathbf{P}^{-1}\mathbf{C}\mathbf{P}$.
Take $\mathbf{P}=\mathbf{B}^{1/2}$ (invertible because $\mathbf{B}\succ\mathbf{0}$).
Then $\mathbf{P}^{-1}=\mathbf{B}^{-1/2}$ and:
\begin{align*}
    \mathbf{P}^{-1}\,\mathbf{S}\,\mathbf{P}
    &= \mathbf{B}^{-1/2}
       \bigl(\mathbf{B}^{-1/2}\mathbf{H}\mathbf{B}^{-1/2}\bigr)
       \mathbf{B}^{1/2} \\
    &= \underbrace{\mathbf{B}^{-1/2}\mathbf{B}^{-1/2}}_{=\mathbf{B}^{-1}}
       \mathbf{H}
       \underbrace{\mathbf{B}^{-1/2}\mathbf{B}^{1/2}}_{=\mathbf{I}} \\
    &= \mathbf{B}^{-1}\mathbf{H}.
\end{align*}
So, $\mathbf{B}^{-1}\mathbf{H}=\mathbf{P}^{-1}\mathbf{S}\mathbf{P}$. $\square$
\end{proof}


Since $\mathbf{S}$ is symmetric and positive definite, its eigenvalues must be real and strictly positive. Therefore, the eigenvalues of $\mathbf{B}^{-1}\mathbf{H}$, which we will call $\mu$, are also real and strictly positive ($\mu > 0$).

\subsection{Step 2: Upper bound of the eigenvalues}

\begin{lemma}[$\mathbf{S}\preceq\mathbf{I} \Leftrightarrow \mathbf{H}\preceq\mathbf{B}$]
\label{app:lem:SpreI}
For symmetric positive definite $\mathbf{B}$ and symmetric $\mathbf{H}$:
\[
    \mathbf{S} = \mathbf{B}^{-1/2}\mathbf{H}\mathbf{B}^{-1/2} \preceq \mathbf{I}
    \iff
    \mathbf{H} \preceq \mathbf{B}.
\]
\end{lemma}
\begin{proof}
$\mathbf{S}\preceq\mathbf{I}$ means $\mathbf{u}^T\mathbf{u}\ge\mathbf{u}^T\mathbf{S}\mathbf{u}$ for all $\mathbf{u}$.
Substitute $\mathbf{u}=\mathbf{B}^{1/2}\mathbf{v}$ (a bijection on $\mathbb{R}^H$ since $\mathbf{B}^{1/2}$ is invertible):
\begin{align*}
    (\mathbf{B}^{1/2}\mathbf{v})^T(\mathbf{B}^{1/2}\mathbf{v})
    &\ge
    (\mathbf{B}^{1/2}\mathbf{v})^T
    \mathbf{B}^{-1/2}\mathbf{H}\mathbf{B}^{-1/2}
    (\mathbf{B}^{1/2}\mathbf{v}) \\[4pt]
    \mathbf{v}^T\underbrace{\mathbf{B}^{1/2}\mathbf{B}^{1/2}}_{=\mathbf{B}}\mathbf{v}
    &\ge
    \mathbf{v}^T\underbrace{\mathbf{B}^{1/2}\mathbf{B}^{-1/2}}_{=\mathbf{I}}\mathbf{H}
    \underbrace{\mathbf{B}^{-1/2}\mathbf{B}^{1/2}}_{=\mathbf{I}}\mathbf{v} \\[4pt]
    \mathbf{v}^T\mathbf{B}\mathbf{v} &\ge \mathbf{v}^T\mathbf{H}\mathbf{v} \quad\forall\,\mathbf{v},
\end{align*}
which is exactly $\mathbf{H}\preceq\mathbf{B}$. $\square$
\end{proof}

It remains to verify $\mathbf{H}\preceq\mathbf{B}$, i.e.\ $\mathbf{B}-\mathbf{H}\succeq\mathbf{0}$.
We have that:
\[
    \mathbf{B}-\mathbf{H}
    = \lambda\,\mathrm{diag}\!\left(\frac{1}{|w^*_i|}\text{ on }\mathcal{A},\ 0\text{ on }\mathcal{P}\right).
\]
This is a diagonal matrix with non-negative entries, so $\mathbf{B}-\mathbf{H}\succeq\mathbf{0}$.
By Lemma~\ref{app:lem:SpreI}, $\mathbf{S}\preceq\mathbf{I}$, so $\mu_i\le 1$ for all $i$.


\subsection{Step 3: Conclusion on the spectral radius}
The eigenvalues of the Jacobian $\mathbf{I} - \mathbf{B}^{-1}\mathbf{H}$ are $1 - \mu_i$.
Because $\mu_i$ is in $(0, 1]$, the value $1 - \mu_i$ must be in $[0, 1)$. Since all eigenvalues are strictly less than 1, the spectral radius is also strictly less than 1:
\begin{equation}
    \rho\bigl(\nabla\mathcal{M}(\mathbf{w}^*)\bigr) < 1
    \label{app:eq:spectral}
\end{equation}

\section{Local Linear Convergence}
\label{app:sec:ostrowski}

\begin{theorem}[Ostrowski's theorem of attraction,~{\cite[Ch.~10]{ortega1970iterative}}]
Let $\mathcal{M}:\mathbb{R}^H\to\mathbb{R}^H$ be of class $C^1$ in a neighbourhood of a
fixed point $\mathbf{w}^*$, and suppose $\rho(\nabla\mathcal{M}(\mathbf{w}^*))<1$.
Then $\mathbf{w}^*$ is a point of attraction: there exists an open ball $B(\mathbf{w}^*,r)$
such that for every $\mathbf{w}_0\in B(\mathbf{w}^*,r)$ the sequence
$\mathbf{w}_{k+1}=\mathcal{M}(\mathbf{w}_k)$ converges to $\mathbf{w}^*$, and the convergence
is \emph{linear} with asymptotic rate $\rho(\nabla\mathcal{M}(\mathbf{w}^*))$:
\[
    \lim_{k\to\infty}\frac{\|\mathbf{w}_{k+1}-\mathbf{w}^*\|}{\|\mathbf{w}_k-\mathbf{w}^*\|}
    = \rho\!\bigl(\nabla\mathcal{M}(\mathbf{w}^*)\bigr).
\]
\end{theorem}
 
We have established $\rho(\nabla\mathcal{M}(\mathbf{w}^*))<1$ in~\eqref{app:eq:spectral},
and $\mathcal{M}$ is $C^1$ at $\mathbf{w}^*$ under the generic assumption that no component
lies on the clip boundary. The hypotheses of Ostrowski's theorem are therefore satisfied,
and IRLS converges locally with linear rate $\rho=1-\mu_{\min}$.

\paragraph{Why the rate is close to 1.}
The factor $\rho=1-\mu_{\min}$ is close to $1$ because the smallest eigenvalue
$\mu_{\min}$ of $\mathbf{B}^{-1}\mathbf{H}$ is small. On the active set the true penalty
$|w_i|$ is affine (zero curvature), so $\mathbf{H}$ carries there only the data curvature,
whereas the surrogate $\mathbf{B}=\mathbf{Q}(\mathbf{w}^*)$ adds $\lambda/|w^*_i|$. A direction
$\mathbf{v}$ concentrated on an active component with small $|w^*_i|$ then has
$\mathbf{v}^T\mathbf{H}\mathbf{v}/\mathbf{v}^T\mathbf{B}\mathbf{v}\to 0$, which drives
$\mu_{\min}\to 0$ and $\rho\to 1$: IRLS is slow precisely where the LASSO penalty is affine.
Section~\ref{sec:convergence} measures $\rho\approx 0.99$, nearly independent of $\varepsilon$.

\section{Convergence Curve Behavior}
\label{app:sec:nonlinear}

When plotting the objective gap on a logarithmic scale, the curve might not appear perfectly linear. This is because three different phases overlap:

\begin{enumerate}
    \item \textbf{Initial transient.} The first iterations close most of the gap very quickly. The Ostrowski theorem only guarantees linear convergence locally near the solution, so this initial phase behaves differently.
    \item \textbf{Linear tail.} Once the algorithm enters the local neighborhood, the linear rate takes over. Because the rate factor is very close to 1 (approximately 0.99), the gap decreases slowly. The curve becomes a straight line on the logarithmic plot.
    \item \textbf{Smoothing floor.} The gap cannot fall below the approximation error introduced by the smoothing threshold. The curve flattens completely when it reaches this artificial limit.
\end{enumerate}

To isolate and confirm the linear convergence, we can track the ratio between consecutive errors relative to the smoothed optimum. For linear convergence, this ratio stabilizes to a constant value.

\chapter{Derivation of the per-step Polyak inequality \texorpdfstring{\eqref{eq:def-step-polyak}}{}}
\label{app:def-step-polyak}

We derive the per-step descent estimate~\eqref{eq:def-step-polyak} of the proof of Theorem~\ref{thm:dsm_convergence} from three ingredients of~\cite{dantonio2009deflected}: the basic identity (2.9), Corollary~3.2, and the Polyak stepsize~\eqref{eq:sr_step}. The notation map between the paper and the report is $\alpha_k^{\text{paper}} \leftrightarrow \gamma_i$, $\beta_k^{\text{paper}} \leftrightarrow \beta_i$, $\nu_k^{\text{paper}} \leftrightarrow \alpha_i$; the exact-oracle regime fixes $\sigma_k = 0$, the uncorrected Polyak variant fixes the inner correction $\gamma_k$ of (3.1) of~\cite{dantonio2009deflected} to zero, and we evaluate the result at $\bar x = \mathbf{w}^*$.

\paragraph{Step 1 --- Identity (2.9).}
In the unconstrained case $X = \R^{H}$ no projection is needed and (2.9) of~\cite{dantonio2009deflected} reduces to expanding the square $\norm{\mathbf{w}_i - \alpha_i \mathbf{d}_i - \mathbf{w}^*}_{2}^{2}$:
\begin{equation}\label{eq:app-prf-29}
\norm{\mathbf{w}_{i+1} - \mathbf{w}^*}_{2}^{2} \;=\; \norm{\mathbf{w}_i - \mathbf{w}^*}_{2}^{2} \;-\; 2\alpha_i\,\langle \mathbf{d}_i, \mathbf{w}_i - \mathbf{w}^* \rangle \;+\; \alpha_i^{2}\,\norm{\mathbf{d}_i}_{2}^{2}.
\end{equation}

\paragraph{Step 2 --- Corollary~3.2 of~\cite{dantonio2009deflected}.}
Hypothesis~(2.13) is verified in step~(a) of the proof of Theorem~\ref{thm:dsm_convergence}, so Corollary~3.2 of~\cite{dantonio2009deflected} (specialised at $\bar x = \mathbf{w}^*$, $\sigma_k = 0$, $\bar\sigma_k = 0$) gives
\[
\langle \mathbf{d}_i,\, \mathbf{w}^* - \mathbf{w}_i \rangle \;\le\; \gamma_i\bigl(f^* - f(\mathbf{w}_i)\bigr) \;+\; \bigl[f(\mathbf{w}^*) - f^* + 0\bigr] \;=\; -\gamma_i\bigl(f(\mathbf{w}_i) - f^*\bigr),
\]
that is,
\begin{equation}\label{eq:app-prf-32}
\langle \mathbf{d}_i,\, \mathbf{w}_i - \mathbf{w}^* \rangle \;\ge\; \gamma_i\,\bigl(f(\mathbf{w}_i) - f^*\bigr).
\end{equation}
The factor $\gamma_i$ on the right-hand side (and not the full $1$ that convexity would give for the plain subgradient $\mathbf{g}_i$) is the deflection-induced loss: $\mathbf{d}_i$ is a convex combination involving past subgradients (those at $\mathbf{w}_{i-1}, \mathbf{w}_{i-2}, \dots$), so only the weight $\gamma_i$ on the current $\mathbf{g}_i$ can be certified to separate $\mathbf{w}_i$ from $\mathbf{w}^*$ by the subgradient inequality.

\paragraph{Step 3 --- Polyak stepsize substitution.}
We work in the regime where Lemma~3.8 of~\cite{dantonio2009deflected} has driven the target $f^{k}_{\mathrm{ref}} - \delta_{k}$ to $f^*$, so~\eqref{eq:sr_step} reads $\alpha_i = \beta_i (f(\mathbf{w}_i) - f^*)/\norm{\mathbf{d}_i}_{2}^{2}$. Substituting into the two non-constant terms of~\eqref{eq:app-prf-29} and applying~\eqref{eq:app-prf-32}:
\begin{align}
-2\alpha_i\,\langle \mathbf{d}_i, \mathbf{w}_i - \mathbf{w}^* \rangle
\;&\overset{\eqref{eq:app-prf-32}}{\le}\; -2\alpha_i\,\gamma_i\bigl(f(\mathbf{w}_i) - f^*\bigr)
\;=\; -2\,\beta_i\,\gamma_i\,\frac{(f(\mathbf{w}_i) - f^*)^{2}}{\norm{\mathbf{d}_i}_{2}^{2}}, \label{eq:app-prf-step3a}\\
\alpha_i^{2}\,\norm{\mathbf{d}_i}_{2}^{2}
\;&=\; \biggl(\frac{\beta_i\,(f(\mathbf{w}_i) - f^*)}{\norm{\mathbf{d}_i}_{2}^{2}}\biggr)^{2}\,\norm{\mathbf{d}_i}_{2}^{2}
\;=\; \beta_i^{2}\,\frac{(f(\mathbf{w}_i) - f^*)^{2}}{\norm{\mathbf{d}_i}_{2}^{2}}. \label{eq:app-prf-step3b}
\end{align}

\paragraph{Step 4 --- Collecting.}
Summing~\eqref{eq:app-prf-step3a} and~\eqref{eq:app-prf-step3b} into~\eqref{eq:app-prf-29},
\[
\norm{\mathbf{w}_{i+1} - \mathbf{w}^*}_{2}^{2}
\;\le\; \norm{\mathbf{w}_i - \mathbf{w}^*}_{2}^{2}
\;-\; \bigl(2\,\beta_i\,\gamma_i \;-\; \beta_i^{2}\bigr)\,\frac{(f(\mathbf{w}_i) - f^*)^{2}}{\norm{\mathbf{d}_i}_{2}^{2}},
\]
and factoring $\beta_i$,
\[
\norm{\mathbf{w}_{i+1} - \mathbf{w}^*}_{2}^{2}
\;\le\; \norm{\mathbf{w}_i - \mathbf{w}^*}_{2}^{2}
\;-\; \beta_i\,(2\gamma_i - \beta_i)\,\frac{(f(\mathbf{w}_i) - f^*)^{2}}{\norm{\mathbf{d}_i}_{2}^{2}},
\]
which is~\eqref{eq:def-step-polyak}. This matches eq.~(3.17) of~\cite{dantonio2009deflected} under the notation map above.

\paragraph{Specialisation to Algorithm~\ref{alg:dsm}.}
Algorithm~\ref{alg:dsm} sets $\beta_i = \min(1, \gamma_i) = \gamma_i$ (because the clip~\eqref{eq:gamma_star} keeps $\gamma_i \le 1$ at every iteration), so the descent factor collapses to
\[
\beta_i\,(2\gamma_i - \beta_i) \;=\; \gamma_i\,(2\gamma_i - \gamma_i) \;=\; \gamma_i^{2} \;\ge\; \gamma_{\min}^{2},
\]
which is the form fed back into the telescoping argument of the proof of Theorem~\ref{thm:dsm_convergence}.

\chapter{Derivation of the optimal deflection \texorpdfstring{$\gamma^{*}$}{gamma*}}
\label{app:gamma-star}

We derive the closed form~\eqref{eq:gamma_star} for the minimiser of
\[
h(\gamma) \;:=\; \norm{\gamma\,\mathbf{g}_i + (1-\gamma)\,\mathbf{d}_{i-1}}_2^{2}
\]
on $\R$, then justify the clip to $[\gamma_{\min}, 1]$. Expanding by bilinearity of the inner product,
\[
h(\gamma) \;=\; \gamma^{2}\,\norm{\mathbf{g}_i}_2^{2} \;+\; 2\,\gamma(1-\gamma)\,\inner{\mathbf{g}_i}{\mathbf{d}_{i-1}} \;+\; (1-\gamma)^{2}\,\norm{\mathbf{d}_{i-1}}_2^{2}.
\]
Collecting powers of $\gamma$,
\begin{equation}\label{eq:app-gamma-parabola}
h(\gamma) \;=\; \underbrace{\bigl(\norm{\mathbf{g}_i}_2^{2} - 2\inner{\mathbf{g}_i}{\mathbf{d}_{i-1}} + \norm{\mathbf{d}_{i-1}}_2^{2}\bigr)}_{=\,\norm{\mathbf{g}_i - \mathbf{d}_{i-1}}_2^{2}}\,\gamma^{2} \;+\; 2\,\bigl(\inner{\mathbf{g}_i}{\mathbf{d}_{i-1}} - \norm{\mathbf{d}_{i-1}}_2^{2}\bigr)\,\gamma \;+\; \norm{\mathbf{d}_{i-1}}_2^{2},
\end{equation}
a parabola in $\gamma$ with leading coefficient $\norm{\mathbf{g}_i - \mathbf{d}_{i-1}}_2^{2} \ge 0$.

\paragraph{Degenerate case $\mathbf{g}_i = \mathbf{d}_{i-1}$.}
The leading coefficient vanishes and $h(\gamma) \equiv \norm{\mathbf{d}_{i-1}}_2^{2}$ is constant on $\R$; moreover the deflected direction $\mathbf{d}_i = \gamma\,\mathbf{g}_i + (1-\gamma)\,\mathbf{d}_{i-1} = \mathbf{d}_{i-1}$ for every $\gamma$, so the algorithm is insensitive to the choice. We set $\gamma_i = 1$ by convention (consistent with the unclipped Polyak first step $\mathbf{d}_0 = \mathbf{g}_0$).

\paragraph{Generic case $\mathbf{g}_i \neq \mathbf{d}_{i-1}$.}
The leading coefficient is strictly positive, so $h$ is strictly convex with a unique unconstrained minimiser. Setting $h^{\prime}(\gamma) = 0$ in~\eqref{eq:app-gamma-parabola},
\[
2\,\norm{\mathbf{g}_i - \mathbf{d}_{i-1}}_2^{2}\,\gamma \;+\; 2\,\bigl(\inner{\mathbf{g}_i}{\mathbf{d}_{i-1}} - \norm{\mathbf{d}_{i-1}}_2^{2}\bigr) \;=\; 0 \quad\Longrightarrow\quad \gamma^{*} \;=\; \frac{\norm{\mathbf{d}_{i-1}}_2^{2} - \inner{\mathbf{g}_i}{\mathbf{d}_{i-1}}}{\norm{\mathbf{g}_i - \mathbf{d}_{i-1}}_2^{2}},
\]
which is~\eqref{eq:gamma_star}.

\paragraph{Projection onto $[\gamma_{\min}, 1]$.}
Since the parabola opens upward, $h$ is decreasing on $(-\infty, \gamma^{*}]$ and increasing on $[\gamma^{*}, +\infty)$. The constrained minimum over $[\gamma_{\min}, 1]$ is therefore
\[
\gamma_i \;=\; \begin{cases}
\gamma_{\min} & \text{if } \gamma^{*} < \gamma_{\min}, \\
\gamma^{*} & \text{if } \gamma^{*} \in [\gamma_{\min}, 1], \\
1 & \text{if } \gamma^{*} > 1,
\end{cases}
\qquad\text{i.e.\ } \gamma_i \;=\; \min\bigl(1,\; \max(\gamma_{\min},\, \gamma^{*})\bigr).
\]
The two clip endpoints serve different purposes: the lower clip $\gamma_{\min}$ enforces the uniform bound $\beta^{*} > 0$ of condition~(3.5) of~\cite{dantonio2009deflected} (Section~\ref{sec:dsm_convergence_conditions}); the upper clip at $1$ keeps $\gamma_i \in (0,1]$ so that $\mathbf{d}_i$ is a convex combination of $\mathbf{g}_i$ and $\mathbf{d}_{i-1}$, a property used in the per-step descent estimate of Appendix~\ref{app:def-step-polyak}.

\paragraph{Interpretation.}
The denominator $\norm{\mathbf{g}_i - \mathbf{d}_{i-1}}_2^{2}$ measures how far the current subgradient is from the previous deflected direction: large values shrink $\gamma^{*}$ and tilt the deflected direction toward memory $\mathbf{d}_{i-1}$. The numerator $\norm{\mathbf{d}_{i-1}}_2^{2} - \inner{\mathbf{g}_i}{\mathbf{d}_{i-1}}$ captures alignment: it is small (so $\gamma^{*}$ is small) when $\mathbf{g}_i$ and $\mathbf{d}_{i-1}$ point in similar directions, and large when they oppose each other.

In the zig-zag regime $\mathbf{g}_i \approx -\mathbf{d}_{i-1}$ with $\norm{\mathbf{g}_i}_2 \approx \norm{\mathbf{d}_{i-1}}_2$, the numerator is approximately $\norm{\mathbf{d}_{i-1}}_2^{2} + \norm{\mathbf{d}_{i-1}}_2^{2} = 2\,\norm{\mathbf{d}_{i-1}}_2^{2}$ while the denominator is approximately $\norm{2\,\mathbf{d}_{i-1}}_2^{2} = 4\,\norm{\mathbf{d}_{i-1}}_2^{2}$, so $\gamma^{*} \approx 1/2$: the deflected direction $\mathbf{d}_i$ splits evenly between $\mathbf{g}_i$ and $\mathbf{d}_{i-1}$ and the oscillatory component cancels.
