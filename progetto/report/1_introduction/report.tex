\section{Problem 25 ML}
\label{sec:intro_problem}
\noindent \textbf{25. (P)} is so-called \textit{extreme learning}, i.e., a neural network
with one hidden layer, $y = w \sigma(W_1 x)$, where the weight matrix for the hidden layer $W_1$
is a fixed random matrix, $\sigma(\cdot)$ is an elementwise activation function of your choice,
and the output weight vector $w$ is chosen by solving a least-squares
problem with $L_1$ regularization $\arg\min_w f(w)$, with $f(w) = \frac{1}{2}\|X w - y \|_2^2 + \lambda \|w\|_1$.

\vspace{0.3cm}

\noindent \textbf{(A1)} is \textit{iteratively reweighted least squares}: i.e., an
iteration where you solve at each step the least-squares problem
$$ w_{k+1} = \arg \min_w f_k(w), \quad f_k(w) = \frac{1}{2} \|X w - y \|_2^2 + \lambda \|W_k w\|_2^2 $$
where $W_k$ is the diagonal matrix with entries $(W_k)_{ii} = |(w_k)_i|^{-1/2}$.

\vspace{0.3cm}
\noindent \textbf{(A2)} is an algorithm of the class of the \textit{deflected subgradient methods}.

\section{Extreme Learning Machine}
An Extreme Learning Machine \cite{HUANG2006489} is a single-hidden-layer feedforward network in which the input-to-hidden weights are fixed at random and only the hidden-to-output weights are trained. We fix the notation: $\mathbf{W}_1 \in \mathbb{R}^{H \times N}$ is the random, frozen input-to-hidden matrix ($H$ hidden units, $N$ inputs); $\sigma(\cdot)$ is an elementwise nonlinearity, taken to be $\tanh$; $\mathbf{w} \in \mathbb{R}^H$ is the trainable output vector; the model output is $\hat{y} = \mathbf{w}^T\sigma(\mathbf{W}_1\mathbf{x}) \in \mathbb{R}$. Since $\mathbf{W}_1$ is fixed, training $\mathbf{w}$ reduces to a (regularized) linear least-squares problem in the hidden-layer features.

\section{Least-Squares Problem}
\label{sec:least_squares}
% The Least-Squares Problem is an optimization problem used in machine learning for data fitting. It aims at finding the best curve that minimizes the sum of squared differences between the \textit{predicted value} $A \mathbf{x}$ and the observed \textit{ground truth} (or \textit{target}) $\mathbf{y}$. Mathematically, we can represent this problem as follows:
% \begin{equation}
%     \min_\mathbf{x} || \mathbf{A} \mathbf{x} - \mathbf{y} ||^2_2
% \end{equation}
Let $\mathbf{X}_\text{origin} \in \mathbb{R}^{M \times N}$ be the matrix of $M$ training inputs stacked as rows, $\mathbf{X} = \sigma(\mathbf{X}_\text{origin} \mathbf{W}_1^T) \in \mathbb{R}^{M \times H}$ the hidden-layer activations, and $\mathbf{y} \in \mathbb{R}^M$ the target vector. Training $\mathbf{w}$ amounts to solving \cite{cortinovis-introleastsquares}
\begin{equation}
\min_{\mathbf{w}} \tfrac{1}{2}\| \mathbf{Xw} - \mathbf{y} \|^2_2,
\end{equation}
which under the full-column-rank assumption on $\mathbf{X}$ admits the unique closed-form solution $\mathbf{w}_0 = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{y}$. Adding the $L_1$ regularization term (Section~\ref{sec:regularization}) breaks this closed form and forces iterative methods.

\section{$L_1$ regularization}
\label{sec:regularization}
The project statement prescribes an $L_1$ penalty on $\mathbf{w}$ \cite{l1-regularization-frangioni}, giving the regularized objective
\begin{equation}
\label{eq:original_problem}
\min_{\mathbf{w}} f(\mathbf{w}) = \frac{1}{2}\| \mathbf{Xw} - \mathbf{y} \|^2_2 + \lambda_{\text{LASSO}} \sum_{i=1}^H |w_i|.
\end{equation}
Here $\lambda_{\text{LASSO}}$ denotes the regularization parameter called $\lambda$ in the project statement (Section~\ref{sec:intro_problem}); a related symbol $\lambda_{\text{IRLS}} = \tfrac{1}{2}\lambda_{\text{LASSO}}$, induced by the IRLS majorization, will be introduced in Chapter~\ref{chap:irls}. The $L_1$ term makes $f$ non-differentiable wherever any $w_i = 0$, ruling out plain gradient methods and motivating the two algorithms developed in the next chapters (IRLS, Chapter~\ref{chap:irls}; deflected subgradient, Chapter~\ref{chap:dsm}).



\section{Objective Function Analysis}
The full objective function we need to minimize is the Lasso regularization of the Least-Squares problem:
$$
\min_{\mathbf{w}} f(\mathbf{w}) = \underbrace{\frac{1}{2}\|\mathbf{Xw} - \mathbf{y}\|_2^2}_{g(\mathbf{w})}
+ \underbrace{\lambda_{\text{LASSO}}\|\mathbf{w}\|_1}_{h(\mathbf{w})}
$$

It is useful to analyse the mathematical properties of $f(\mathbf{w})$, as they
determine which algorithms can be applied.

\subsection{Convexity}
$f$ is the sum of two convex functions \cite{convex-functions-frangioni}. The quadratic term $g(\mathbf{w}) = \frac{1}{2}\|\mathbf{Xw} - \mathbf{y}\|_2^2$ has Hessian $\mathbf{X}^T\mathbf{X} \succeq 0$, so it is convex; under the ELM full-column-rank assumption the Hessian is positive definite and $g$ is in fact \emph{strictly} convex. The $\ell_1$ penalty $h(\mathbf{w}) = \lambda_{\text{LASSO}}\|\mathbf{w}\|_1$ is convex as a scaled norm, but it is \emph{not} strictly convex (the segment between two sign-aligned vectors keeps it affine). Strict convexity of $f = g + h$ therefore relies entirely on $g$: whenever $g$ is strictly convex, so is $f$, and the minimizer $\mathbf{w}^*$ is unique.

\begin{proposition}
    \label{prop:irls_prop}
    If $\mathbf{X} \in \mathbb{R}^{M \times H}$ has full column rank, then $f(\mathbf{w}) = \frac{1}{2}\|\mathbf{Xw} - \mathbf{y}\|_2^2 + \lambda_{\text{LASSO}}\|\mathbf{w}\|_1$ is strictly convex and admits a unique global minimizer $\mathbf{w}^*$.
\end{proposition}

\subsection{Non-smoothness}
Unlike the quadratic term, the $L_1$ penalty $h(\mathbf{w}) = \lambda_{\text{LASSO}}\|\mathbf{w}\|_1$
is \textbf{not differentiable} \cite{nondifferentiability-l1-frangioni} at any point where some component $w_i = 0$. This means
$f$ itself is non-smooth, and we cannot rely on gradient-based methods alone.
At points where all $w_i \neq 0$, the gradient exists and is:
$$
\nabla f(\mathbf{w}) = \mathbf{X}^T(\mathbf{Xw} - \mathbf{y}) + \lambda_{\text{LASSO}} \operatorname{sign}(\mathbf{w})
$$
where $\operatorname{sign}(\mathbf{w})$ is applied element-wise.
At points where some $w_i = 0$ one must reason in terms of \textit{subgradients}, which we recall formally in the next subsection and use extensively in Chapter~\ref{chap:dsm}.

\subsection{Optimality conditions}
\label{sec:intro_optimality}
Since $f$ is non-smooth, the classical condition $\nabla f(\mathbf{w}^*) = 0$
does not apply at every minimizer. Instead, the optimality condition is expressed
via the \textbf{subdifferential} \cite{subdifferential-introduction-frangioni}: $\mathbf{w}^*$ is a global minimizer for our ELM problem if and
only if:
$$
0 \in \partial f(\mathbf{w}^*) = \mathbf{X}^T(\mathbf{Xw}^* - \mathbf{y})
+ \lambda_{\text{LASSO}} \, \partial\|\mathbf{w}^*\|_1
$$
where the subdifferential of the $L_1$ norm is given component-wise by:
$$
[\partial \|\mathbf{w}\|_1]_i =
\begin{cases}
    \{+1\}    & \text{if } w_i > 0,  \\
    \{-1\}    & \text{if } w_i < 0,  \\
    [-1, +1]  & \text{if } w_i = 0.
\end{cases}
$$
This condition can be written component-wise as: for each $i = 1, \ldots, H$,
\begin{equation}
\label{eq:optimality}
\left[\mathbf{X}^T(\mathbf{Xw}^* - \mathbf{y})\right]_i + \lambda_{\text{LASSO}} s_i = 0,
\qquad s_i \in \partial|w_i^*|.
\end{equation}
Intuitively, $\lambda_{\text{LASSO}}$ controls the sparsity of $\mathbf{w}^*$: the larger
$\lambda_{\text{LASSO}}$ is, the easier it is for the optimality condition to force components
$w_i^* = 0$, producing a sparser solution.
