\section{Problem 25 ML}
\noindent \textbf{25. (P)} is so-called \textit{extreme learning}, i.e., a neural network
with one hidden layer, $y = w \sigma(W_1 x)$, where the weight matrix for the hidden layer $W_1$
is a fixed random matrix, $\sigma(\cdot)$ is an elementwise activation function of your choice,
and the output weight vector $w$ is chosen by solving a least-squares
problem with $L_1$ regularization $\arg\min_w f(w)$, with $f(w) = \frac{1}{2}\|X w - y \|_2^2 + \lambda \|w\|_1$.

\vspace{0.3cm}

\noindent \textbf{(A1)} is \textit{iteratively reweighted least squares}: i.e., an
iteration where you solve at each step the least-squares problem
$$ w_{k+1} = \arg \min_w f_k(w), \quad f_k(w) = \frac{1}{2} \|X w - y \|_2^2 + \lambda \|W_k w\|_2^2 $$
where $W_k$ is the diagonal matrix with entries $(W_k)_{ii} = |(w_k)_i|^{-1/2}$.

\vspace{0.3cm}
\noindent \textbf{(A2)} is an algorithm of the class of the \textit{deflected subgradient methods}.

\section{Extreme Learning Machine}
Extreme Learning Machine (ELM) \cite{HUANG2006489} is a machine learning model implemented as a neural network with a single hidden layer, in which:
\begin{itemize}
    \item $\mathbf{W_1} \in \mathbb{R}^{H \times N}$ is the input-to-hidden weight matrix (where $H$ is the number of nodes in the hidden layer and $N$ is the number of input nodes) that is \textbf{randomly generated} and \textbf{static} (does not change during training) \cite{HUANG2006489}.
    \item $\sigma(\cdot)$ is the element-wise (nonlinear) activation function applied to the $\mathbf{W_1x} \in \mathbb{R}^H$ vector, where $\textbf{x} \in \mathbb{R}^N$ is the input vector. The choice of using an activation function for the hidden layer results, under mild assumptions and with high probability, in the hidden layer feature matrix with full rank when $M \ge H$, since random projections through a nonlinear function produce linearly independent features.
    \item $\mathbf{w} \in \mathbb{R}^{H}$ is the hidden-to-output weight vector, chosen by solving the least-squares problem.
    \item $\hat{y} = \mathbf{w}^T\sigma(\mathbf{W_1x}) \in \mathbb{R}$ is the prediction of the model.
\end{itemize}

The key insight behind this architecture is that randomly fixing $\mathbf{W_1}$ transforms an extremely hard non-linear optimization problem (training a full neural network) into a simpler \textbf{linear system} resolution. Classical neural networks must learn \textit{all} weights jointly, which requires non-convex optimization with no guarantee of finding the global minimum \cite{goodfellow2016deep}. In an ELM, the only free parameters are the output weights $\mathbf{w}$, and once $\mathbf{W_1}$ is fixed, the hidden-layer activations $\mathbf{X} = \sigma(\mathbf{X}_\text{origin} \mathbf{W}_1^T)$ are constant. Finding $\mathbf{w}$ then reduces to solving the Least-Squares problem, which is convex with a closed-form solution. This choice works very well in practice: random projections through a non-linear activation function are sufficient to produce rich, expressive features \cite{rahimi2007random} from which a linear model can learn effectively, exploiting the same properties of expressiveness as the \textbf{linear basis expansions} \cite{hastie2009elements} in classical Machine Learning models (e.g. linear models).

\section{Least-Squares Problem}
% The Least-Squares Problem is an optimization problem used in machine learning for data fitting. It aims at finding the best curve that minimizes the sum of squared differences between the \textit{predicted value} $A \mathbf{x}$ and the observed \textit{ground truth} (or \textit{target}) $\mathbf{y}$. Mathematically, we can represent this problem as follows:
% \begin{equation}
%     \min_\mathbf{x} || \mathbf{A} \mathbf{x} - \mathbf{y} ||^2_2
% \end{equation}
Considering $\mathbf{X} = \sigma(\mathbf{X}_\text{origin} \mathbf{W}_1^T) \in \mathbb{R}^{M \times H}$ the matrix of activations of the hidden layer, while $\mathbf{X}_\text{origin} \in \mathbb{R}^{M \times N}$ is the matrix of all the inputs, and $\mathbf{y} \in \mathbb{R}^M$ is the vector of all the target values, the definition of the Least Squares Problem \cite{cortinovis-introleastsquares} in the ELM case is the following:
\begin{equation}
\min_{\mathbf{w}} || \mathbf{Xw} - \mathbf{y} ||^2_2
\end{equation}

The Least-Squares problem is proven to have a unique solution if and only if the matrix $A$ has full column rank. In that case, we can write the optimization problem as follows:
\begin{equation}
\min_\mathbf{w} \frac{1}{2}\mathbf{w}^T\mathbf{X}^T\mathbf{X}\mathbf{w} - \mathbf{y}^T\mathbf{X}\mathbf{w} + \frac{1}{2}\mathbf{y}^T\mathbf{y}
\end{equation}
since the gradient of this function is $\mathbf{X}^T\mathbf{X}\mathbf{w} - \mathbf{X}^T\mathbf{y}$, the Hessian matrix is $\mathbf{X}^T\mathbf{X} \succ 0$, which means that the function is strictly convex. In this case, the minimum exists and is unique, and can be found solving the equation:
$$
\mathbf{X}^T\mathbf{X}\mathbf{w} - \mathbf{X}^T\mathbf{y} = 0
$$
or
$$
\mathbf{X}^T\mathbf{X}\mathbf{w} = \mathbf{X}^T\mathbf{y}
$$
where $\mathbf{X}^T\mathbf{X}$ is square invertible since it's positive definite.   The linear system has a unique solution $\mathbf{w}_0 = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{y}$.
However, adding the $L_1$ regularization term (Section~\ref{sec:regularization}) destroys the closed-form structure, requiring iterative methods.

\section{Regularization Techniques}
\label{sec:regularization}
% When solving the Least-Squares Problem, we may prefer solutions that use a subset of available features. This prevents the problem of overfitting the model to the data, and producing a result that is more robust with respect to the data given. Regularization is the technique used to add a penalty to weights, and control the free parameters of the model to avoid overfitting to data.


\subsection{$L_1$ regularization}
The $L_1$ regularization \cite{l1-regularization-frangioni} is the best convex approximation for the $L_0$ norm, which penalizes the number of non-zero weights. Because of this sparsity, we are going to use the $L_1$ instead of $L_2$, which shrinks the weights but rarely zeroes them out.

When adding the $L_1$ regularization term to the Least-Squares problem in the case of ELM, the equation is the following:
\begin{equation}
\label{eq:original_problem}
\min_{\mathbf{w}} f(\mathbf{w}) = \frac{1}{2}|| \mathbf{Xw} - \mathbf{y} ||^2_2 + \lambda_{\text{LASSO}} \sum_i^H|w_i|
\end{equation}

The price to pay is the non-differentiability of $f(\mathbf{w})$ at any point where some $w_i = 0$, which prevents direct use of gradient-based methods.

Because of that, we can use the Deflected Subgradient Methods (\ref{alg:dsm}) and the IRLS algorithm (\ref{chap:irls}), which we will see in the following chapters.



\section{Objective Function Analysis}
The full objective function we need to minimize is the Lasso regularization of the Least-Squares problem:
$$
\min_{\mathbf{w}} f(\mathbf{w}) = \underbrace{\frac{1}{2}\|\mathbf{Xw} - \mathbf{y}\|_2^2}_{g(\mathbf{w})}
+ \underbrace{\lambda_{\text{LASSO}}\|\mathbf{w}\|_1}_{h(\mathbf{w})}
$$

It is useful to analyse the mathematical properties of $f(\mathbf{w})$, as they
determine which algorithms can be applied.

\subsection{Convexity}
$f(\mathbf{w})$ is the sum of two convex functions \cite{Boyd_Vandenberghe_2004}. The quadratic term
$g(\mathbf{w}) = \frac{1}{2}\|\mathbf{Xw} - \mathbf{y}\|_2^2$ has Hessian
$\nabla^2 g(\mathbf{w}) = \mathbf{X}^T\mathbf{X} \succeq 0$, making it convex.
If $\mathbf{X}$ has full column rank, then $\mathbf{X}^T\mathbf{X} \succ 0$ and
$g$ is \textit{strictly convex}. The regularization term
$h(\mathbf{w}) = \lambda_{\text{LASSO}}\|\mathbf{w}\|_1$ is convex as a scaled norm.
Therefore $f = g + h$ is convex; and if $\mathbf{X}$ has full column rank,
$f$ is \textbf{strictly convex} and admits a \textbf{unique global minimum}.

\begin{proposition}
    \label{prop:irls_prop}
    If $\mathbf{X} \in \mathbb{R}^{M \times H}$ has full column rank, then
    $f(\mathbf{w}) = \frac{1}{2}\|\mathbf{Xw} - \mathbf{y}\|_2^2 + \lambda_{\text{LASSO}}\|\mathbf{w}\|_1$
    is strictly convex and admits a unique global minimizer $\mathbf{w}^*$ \cite{Boyd_Vandenberghe_2004}.
\end{proposition}
\begin{proof}
We write $f = g + h$ where $g(\mathbf{w}) = \frac{1}{2}\|\mathbf{Xw} - \mathbf{y}\|_2^2$
and $h(\mathbf{w}) = \lambda_{\text{LASSO}}\|\mathbf{w}\|_1$.
 
\textbf{Strict convexity.}
Since $\mathbf{X}$ has full column rank, for any $\mathbf{v} \neq \mathbf{0}$ we have
$\mathbf{Xv} \neq \mathbf{0}$, so $\mathbf{v}^T(\mathbf{X}^T\mathbf{X})\mathbf{v}
= \|\mathbf{Xv}\|_2^2 > 0$. Hence the Hessian of $g$ is
$\nabla^2 g = \mathbf{X}^T\mathbf{X} \succ 0$, making $g$ strictly convex.
$h$ is convex as a non-negative multiple of a norm. The sum of a strictly convex
function and a convex function is strictly convex \cite{convex-functions-frangioni}, so $f$ is strictly convex.
 
\textbf{Existence and uniqueness of the minimizer.}
Since $\mathbf{X}^T\mathbf{X} \succ 0$, $g$ is coercive:
$g(\mathbf{w}) \to +\infty$ as $\|\mathbf{w}\| \to \infty$. Because $h \geq 0$,
$f = g + h$ is also coercive. This means the sub-level set
$\{\mathbf{w} : f(\mathbf{w}) \leq f(\mathbf{0})\}$ is compact, and since
$f$ is continuous, it attains its minimum on this set. Strict convexity then
rules out two distinct minimizers: if $\mathbf{w}_1 \neq \mathbf{w}_2$ both
minimized $f$, the midpoint $\frac{\mathbf{w}_1 + \mathbf{w}_2}{2}$ would give
a strictly smaller value, a contradiction.
\end{proof}

\subsection{Non-smoothness}
Unlike the quadratic term, the $L_1$ penalty $h(\mathbf{w}) = \lambda_{\text{LASSO}}\|\mathbf{w}\|_1$
is \textbf{not differentiable} \cite{nondifferentiability-l1-frangioni} at any point where some component $w_i = 0$. This means
$f$ itself is non-smooth, and we cannot rely on gradient-based methods alone.
At points where all $w_i \neq 0$, the gradient exists and is:
$$
\nabla f(\mathbf{w}) = \mathbf{X}^T(\mathbf{Xw} - \mathbf{y}) + \lambda_{\text{LASSO}} \operatorname{sign}(\mathbf{w})
$$
where $\operatorname{sign}(\mathbf{w})$ is applied element-wise.
At points where some $w_i = 0$, one must reason in terms of
\textit{subgradients} (see Chapter~\ref{alg:dsm}).

\subsection{Optimality conditions}
\label{sec:intro_optimality}
Since $f$ is non-smooth, the classical condition $\nabla f(\mathbf{w}^*) = 0$
does not apply at every minimizer. Instead, the optimality condition is expressed
via the \textbf{subdifferential} \cite{subdifferential-introduction-frangioni}: $\mathbf{w}^*$ is a global minimizer for our ELM problem if and
only if:
$$
0 \in \partial f(\mathbf{w}^*) = \mathbf{X}^T(\mathbf{Xw}^* - \mathbf{y})
+ \lambda_{\text{LASSO}} \, \partial\|\mathbf{w}^*\|_1
$$
where the subdifferential of the $L_1$ norm is given component-wise by:
$$
[\partial \|\mathbf{w}\|_1]_i =
\begin{cases}
    \{+1\}    & \text{if } w_i > 0,  \\
    \{-1\}    & \text{if } w_i < 0,  \\
    [-1, +1]  & \text{if } w_i = 0.
\end{cases}
$$
This condition can be written component-wise as: for each $i = 1, \ldots, H$,
\begin{equation}
\label{eq:optimality}
\left[\mathbf{X}^T(\mathbf{Xw}^* - \mathbf{y})\right]_i + \lambda_{\text{LASSO}} s_i = 0,
\qquad s_i \in \partial|w_i^*|.
\end{equation}
Intuitively, $\lambda_{\text{LASSO}}$ controls the sparsity of $\mathbf{w}^*$: the larger
$\lambda_{\text{LASSO}}$ is, the easier it is for the optimality condition to force components
$w_i^* = 0$, producing a sparser solution.
